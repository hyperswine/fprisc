cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET esp32c3)
set(COMPONENTS main)
set(FPR_PROGRAM "${CMAKE_CURRENT_SOURCE_DIR}/examples/echo.fpr" CACHE FILEPATH "Builtin entry point")
option(FPR_FLASH_BOOT "Build a bootable application for an ESP-IDF app partition" OFF)
option(FPR_FLASH_FILES "Include the SPIFFS file adapter" OFF)
set(FPR_FLASH_LABEL "fprfiles" CACHE STRING "Existing data/SPIFFS partition to mount")
# IDF discovers component dependencies in a separate cmake -P process.
set(ENV{FPR_IDF_FLASH_FILES} 0)
if(FPR_FLASH_BOOT)
    set(SDKCONFIG_DEFAULTS "sdkconfig.flash")
else()
    set(SDKCONFIG_DEFAULTS "sdkconfig.defaults")
endif()
if(FPR_FLASH_FILES)
    set(ENV{FPR_IDF_FLASH_FILES} 1)
    list(APPEND SDKCONFIG_DEFAULTS "sdkconfig.files")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(c3_builtin)
if((FPR_FLASH_BOOT AND CONFIG_APP_BUILD_TYPE_RAM) OR (NOT FPR_FLASH_BOOT AND NOT CONFIG_APP_BUILD_TYPE_RAM))
    message(FATAL_ERROR "Boot mode and sdkconfig disagree; select a separate SDKCONFIG for each boot mode")
endif()
