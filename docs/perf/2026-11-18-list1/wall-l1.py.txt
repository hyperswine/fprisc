import os, sys, time, subprocess, json, shutil, statistics
A="/workspace/out/apps"; L="/workspace/l1"; R=int(sys.argv[1]); OUT=sys.argv[2]
env=dict(os.environ, LD_LIBRARY_PATH="/opt/tc/fprdeps/lib")
def fresh(src,name):
    d=f"{L}/run/wall-{name}"; os.makedirs(d,exist_ok=True); shutil.copy(src,f"{d}/x.log"); return f"{d}/x.log"
jobs=[]
for v in ["base","proto","protoB"]:
    jobs.append((f"inv-member-{v}", lambda v=v:[f"{L}/inv-{v}/build/inventory","--store="+fresh(f"{A}/data/inv-big.log",f"im-{v}"),"--member=bench","--password=benchpassword1"]))
    jobs.append((f"inv-import-{v}", lambda v=v:[f"{L}/inv-{v}/build/inventory","--store="+fresh(f"{A}/data/inv-big.log",f"ii-{v}"),f"--import={A}/data/firestore-300.json"]))
for v in ["base","proto"]:
    jobs.append((f"p1d-{v}", lambda v=v:[f"{L}/p1-{v}/drv-bin",fresh(f"{A}/data/pos1-big.log",f"pd-{v}")]))
res={}
for r in range(R):
    for name,argv in jobs:
        a=argv(); t0=time.perf_counter()
        p=subprocess.Popen(a,stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL,env=env)
        _,st,ru=os.wait4(p.pid,0); w=time.perf_counter()-t0
        res.setdefault(name,[]).append({"wall":w,"maxrss_mib":ru.ru_maxrss/1024,"rc":os.waitstatus_to_exitcode(st)})
    print("round",r+1,flush=True)
json.dump(res,open(OUT,"w"),indent=1)
for k,v in res.items(): print(k,"wall med %.3f"%statistics.median(x["wall"] for x in v),"rss med %.1f"%statistics.median(x["maxrss_mib"] for x in v),"rc",set(x["rc"] for x in v))
