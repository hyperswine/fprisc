# wall2.py ROUNDS OUT: interleaved wall/maxrss (os.wait4), app PR changes vs base, main f5a6bd9d
import os, sys, time, subprocess, json, shutil, statistics
A="/workspace/out/apps"; L="/workspace/l1"; R=int(sys.argv[1]); OUT=sys.argv[2]
env=dict(os.environ, LD_LIBRARY_PATH="/opt/tc/fprdeps/lib")
def fresh(src,name):
    d=f"{L}/run/w2-{name}"; os.makedirs(d,exist_ok=True); shutil.copy(src,f"{d}/x.log"); return f"{d}/x.log"
big=f"{A}/data/inv-big.log"; jobs=[]
for v in ["base","new"]:
    jobs.append((f"inv-member-{v}",None,lambda v=v:[f"{L}/inv-{v}/build/inventory","--store="+fresh(big,f"im-{v}"),"--member=bench","--password=benchpassword1"]))
    jobs.append((f"inv-import-{v}",None,lambda v=v:[f"{L}/inv-{v}/build/inventory","--store="+fresh(big,f"ii-{v}"),f"--import={A}/data/firestore-300.json"]))
    jobs.append((f"inv-sugg-{v}",None,lambda v=v:[f"{L}/inv-{v}/build/sugg",big]))
    jobs.append((f"inv-live-{v}",None,lambda v=v:[f"{L}/inv-{v}/build/live",big]))
    jobs.append((f"terra-{v}",f"{L}/terra-{v}",lambda v=v:["./drive-bin","300"]))
    jobs.append((f"qrepo-{v}",None,lambda v=v:[f"{L}/q-{v}/bench-bin"]))
for v in ["mbase","mnew","mvec"]:
    jobs.append((f"pos1-{v}",None,lambda v=v:[f"{L}/p1-{v}/drv-bin",fresh(f"{A}/data/pos1-big.log",f"p1-{v}")]))
res={}
for r in range(R):
    for name,cwd,argv in jobs:
        a=argv(); t0=time.perf_counter()
        p=subprocess.Popen(a,cwd=cwd,stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL,env=env)
        _,st,ru=os.wait4(p.pid,0); w=time.perf_counter()-t0
        res.setdefault(name,[]).append({"wall":w,"maxrss_mib":ru.ru_maxrss/1024,"rc":os.waitstatus_to_exitcode(st)})
    print("round",r+1,flush=True)
json.dump(res,open(OUT,"w"),indent=1)
for k,v in res.items(): print(k,"wall med %.3f"%statistics.median(x["wall"] for x in v),"rss med %.1f"%statistics.median(x["maxrss_mib"] for x in v),"rc",sorted(set(x["rc"] for x in v)))
