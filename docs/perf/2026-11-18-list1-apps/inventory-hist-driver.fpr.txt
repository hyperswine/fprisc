# CHECK (outside the repo): the item history text, read from the journal on disk
App = use "../src/inventory".
Program = use "std/program".
List = use "std/list".
main = case Program.args Unit of
    path :: r ->
      m = { App.initModel | path = path };
      _ = List.map (fn id -> print "== {id}\n{App.historyText m id}") ([0 - 1, 0] + List.range 1 60 + [3999, 4000, 4001, 5000]);
      0
  | Nil -> Program.fail 2 "usage: hist LOG".
