import sys,subprocess,re,collections,bisect
from cgaddr import parse
binp,cgf=sys.argv[1],sys.argv[2]; topn=int(sys.argv[3]) if len(sys.argv)>3 else 6
cost=parse(cgf); total=sum(cost.values())
syms=[]
for l in subprocess.run(['nm','-n',binp],capture_output=True,text=True).stdout.splitlines():
    p=l.split()
    if len(p)==3 and p[1] in 'tTwW': syms.append((int(p[0],16),p[2]))
addrs=[s[0] for s in syms]
def sym(a):
    i=bisect.bisect_right(addrs,a)-1
    return syms[i][1] if i>=0 else '?'
dis={}
cur=None
for l in subprocess.run(['objdump','-d','--no-show-raw-insn',binp],capture_output=True,text=True).stdout.splitlines():
    m=re.match(r'\s*([0-9a-f]+):\s+(.*)',l)
    if m: dis[int(m.group(1),16)]=m.group(2).strip()
bysym=collections.Counter()
for a,c in cost.items(): bysym[sym(a)]+=c
print(f"total {total/1e9:.3f} G")
addr_sorted=sorted(dis)
def cls(a,ins,prev):
    i=ins
    if '%fs:' in i: return 'fuel/hart-tls'
    if re.match(r'sub\s+\$0x[0-9a-f]+,\(%r10\)',i): return 'fuel/hart-tls'
    if i.startswith('jg') and prev and re.match(r'sub\s+\$0x[0-9a-f]+,\(%r10\)',prev): return 'fuel/hart-tls'
    if re.match(r'(jno|jo)\b',i): return 'overflow-check'
    if re.match(r'sar\s+\$?0?x?1?,?%',i) or re.match(r'sar\s+%',i) or i.startswith('sar    %'): return 'tag/untag'
    if re.match(r'(sar|shl|add)\s+%r\w+$',i) : return 'tag/untag'
    if re.match(r'lea\s+0x1\(%r\w+,%r\w+,1\)',i) : return 'tag/untag'
    if re.match(r'(and|test)\s+\$0x1,',i): return 'tag/untag'
    if re.match(r'(sar|shl)\s+\$0x1,',i): return 'tag/untag'
    if re.match(r'(mov|lea)\w*\s+.*-?0x[0-9a-f]+\(%rbp\)',i) or re.match(r'mov\w*\s+.*\(%rsp\)',i): return 'spill/frame'
    if i.startswith(('call','ret','push','pop')): return 'call/ret'
    if i.startswith('j'): return 'branch'
    if i.startswith('cmp') or i.startswith('test') or i.startswith('set') or i.startswith('movzb'): return 'compare'
    if re.match(r'mov\w*\s+-?(0x[0-9a-f]+)?\(%',i): return 'load'
    if re.match(r'mov\w*\s+[^,]+,\s*-?(0x[0-9a-f]+)?\(%',i): return 'store'
    if re.match(r'mov\w*\s+%\w+,%\w+$',i): return 'reg-move'
    return 'alu/other'
for s,c in bysym.most_common(topn):
    cc=collections.Counter()
    for a,v in cost.items():
        if sym(a)!=s: continue
        ins=dis.get(a,'?')
        j=bisect.bisect_left(addr_sorted,a); prev=dis.get(addr_sorted[j-1]) if j>0 else None
        cc[cls(a,ins,prev)]+=v
    print(f"\n{s[:90]}  {c/1e9:.3f} G ({100*c/total:.1f}%)")
    print('   '+', '.join(f"{k} {100*v/c:.0f}%" for k,v in cc.most_common()))
